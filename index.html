-- ============================================================ -- Refill needed: items below their Min Stock threshold, with -- filters and order-placed status. Run in Supabase SQL Editor. -- ============================================================ create or replace function get_refill_needed( p_warehouse_id uuid, p_med_types text[] default null, p_min_unique_cust_60d int default null, p_min_bills_60d int default null ) returns table( item_id uuid, item_name text, pack_size text, current_stock numeric, min_stock numeric, max_stock numeric, reorder_qty numeric, medicine_type text, unique_cust_60d integer, bills_60d integer, last_order_date timestamptz, last_order_distributor text, order_placed boolean, purchased_since_order boolean ) language sql stable as $$ with latest_stock as ( select distinct on (item_id) item_id, (coalesce(stock_loose,0) + coalesce(closing_strip,0)) as current_stock, snapshot_at from inventory_snapshots where warehouse_id = p_warehouse_id order by item_id, snapshot_at desc ), latest_analysis as ( select distinct on (item_id) item_id, min_stock, max_stock, reorder_qty, medicine_type, unique_cust_60d, bills_60d from medicine_analysis_snapshots where warehouse_id = p_warehouse_id order by item_id, snapshot_date desc ), last_order as ( select distinct on (ori.item_id) ori.item_id, r.run_at, d.name as distributor_name from order_run_items ori join order_runs r on r.id = ori.order_run_id left join distributors d on d.id = ori.distributor_id where r.warehouse_id = p_warehouse_id order by ori.item_id, r.run_at desc ) select ls.item_id, i.item_name, i.pack_size, ls.current_stock, la.min_stock, la.max_stock, la.reorder_qty, la.medicine_type, la.unique_cust_60d, la.bills_60d, lo.run_at as last_order_date, lo.distributor_name as last_order_distributor, (lo.run_at is not null) as order_placed, coalesce(exists( select 1 from purchase_history ph where ph.item_id = ls.item_id and ph.warehouse_id = p_warehouse_id and lo.run_at is not null and ph.purchase_date >= lo.run_at::date ), false) as purchased_since_order from latest_stock ls join items i on i.id = ls.item_id left join latest_analysis la on la.item_id = ls.item_id left join last_order lo on lo.item_id = ls.item_id where ls.current_stock < coalesce(la.min_stock, 999999999) and (p_med_types is null or la.medicine_type = any(p_med_types)) and (p_min_unique_cust_60d is null or la.unique_cust_60d >= p_min_unique_cust_60d) and (p_min_bills_60d is null or la.bills_60d >= p_min_bills_60d) order by (ls.current_stock = 0) desc, la.unique_cust_60d desc nulls last; $$; grant execute on function get_refill_needed(uuid, text[], int, int) to authenticated;
